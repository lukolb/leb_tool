% Abstände für den RatingHeader.
% Der automatische Kopf nach einem longtable-Seitenumbruch nutzt weiterhin
% \RatingHeaderAutoAfterSpace. Ein expliziter \SkillSectionPageBreak startet
% dagegen eine neue Tabelle und kann deshalb eine andere erste Kopfposition nutzen.

\newcommand{\SkillsTableExplicitTopSpace}{-5mm}

% Abstand nach dem allerersten RatingHeader der Skill-Tabelle
\newcommand{\RatingHeaderFirstAfterSpace}{-13mm}

% Abstand nach RatingHeader bei automatischen longtable-Seitenumbrüchen
\newcommand{\RatingHeaderAutoAfterSpace}{0mm}

% Abstand nach RatingHeader nach explizitem \SkillSectionPageBreak
\newcommand{\RatingHeaderExplicitAfterSpace}{-13mm}

\setlength{\LTright}{0mm}

}

\endhead
